/**

 *
 * <p>FS references and string references are positive 1-based ints. Ref 0 is null.

 * integer/float/string/FS list families are represented as regular FS records

 *
 * <p>Compile with: javac --release 21 --enable-preview ...</p>

    private FfmFeatureStructureBackend(Arena arena,
                                       boolean ownsArena,
                                       long fsHeapBytes,
                                       long featureHeapBytes,
                                       long arrayHeapBytes,
                                       long stringRecordHeapBytes,
                                       long stringByteHeapBytes) {
        this.arena = Objects.requireNonNull(arena, "arena");
        this.ownsArena = ownsArena;
        this.fsHeap = new SegmentHeap(arena, "fs", Math.max(fsHeapBytes, Layouts.FS_RECORD_SIZE), 8);
        this.featureHeap = new SegmentHeap(arena, "features", Math.max(featureHeapBytes, Layouts.SLOT_WIDTH), 8);
        this.arrayHeap = new SegmentHeap(arena, "arrays", Math.max(arrayHeapBytes, Layouts.ARRAY_HEADER_SIZE), 8);
        this.stringRecordHeap = new SegmentHeap(arena, "string-records", Math.max(stringRecordHeapBytes, Layouts.STRING_RECORD_SIZE), 8);
        this.stringByteHeap = new SegmentHeap(arena, "string-bytes", Math.max(stringByteHeapBytes, 1), 1);
        this.indexes = new IndexRepository(arena);
        this.stringsByRef.add(null);
    }

    public synchronized int createFeatureStructure(int typeCode, int featureCount) {
        requireOpen();

        if (typeCode <= 0) {
            throw new IllegalArgumentException("typeCode must be positive");
        }
        if (featureCount < 0) {
            throw new IllegalArgumentException("featureCount must be non-negative");
        }

        long recordOffset = fsHeap.allocate(Layouts.FS_RECORD_SIZE, 8);
        int fsRef = Math.toIntExact(recordOffset / Layouts.FS_RECORD_SIZE) + 1;
        fsCount = Math.max(fsCount, fsRef);

        long featureOffset = featureCount == 0
                ? NULL_OFFSET
                : featureHeap.allocate(featureCount * (long) Layouts.SLOT_WIDTH, 8);

        if (featureOffset != NULL_OFFSET) {
            for (int i = 0; i < featureCount; i++) {
                featureHeap.segment().set(Layouts.I64, featureOffset + i * (long) Layouts.SLOT_WIDTH, 0L);
            }
        }

        MemorySegment fs = fsHeap.segment();
        fs.set(Layouts.I32, recordOffset + Layouts.FS_TYPE_CODE, typeCode);
        fs.set(Layouts.I32, recordOffset + Layouts.FS_FLAGS, 0);
        fs.set(Layouts.I32, recordOffset + Layouts.FS_FEATURE_COUNT, featureCount);
        fs.set(Layouts.I32, recordOffset + Layouts.FS_RESERVED, 0);
        fs.set(Layouts.I64, recordOffset + Layouts.FS_FEATURE_OFFSET, featureOffset);
        fs.set(Layouts.I64, recordOffset + Layouts.FS_AUX_OFFSET, NULL_OFFSET);

        return fsRef;
    }

    public synchronized int putString(String value) {
        requireOpen();

        if (value == null) {
            return 0;
        }

        Integer existing = internedStrings.get(value);
        if (existing != null) {
            return existing;
        }

        byte[] bytes = value.getBytes(StandardCharsets.UTF_8);
        long byteOffset = stringByteHeap.allocate(bytes.length, 1);
        MemorySegment bytesSegment = stringByteHeap.segment();

        for (int i = 0; i < bytes.length; i++) {
            bytesSegment.set(Layouts.I8, byteOffset + i, bytes[i]);
        }

        long recordOffset = stringRecordHeap.allocate(Layouts.STRING_RECORD_SIZE, 8);
        int stringRef = Math.toIntExact(recordOffset / Layouts.STRING_RECORD_SIZE) + 1;

        MemorySegment record = stringRecordHeap.segment();
        record.set(Layouts.I64, recordOffset + Layouts.STRING_BYTE_OFFSET, byteOffset);
        record.set(Layouts.I32, recordOffset + Layouts.STRING_BYTE_LENGTH, bytes.length);
        record.set(Layouts.I32, recordOffset + Layouts.STRING_HASH, value.hashCode());

        stringCount = Math.max(stringCount, stringRef);
        internedStrings.put(value, stringRef);

        while (stringsByRef.size() <= stringRef) {
            stringsByRef.add(null);
        }
        stringsByRef.set(stringRef, value);

        return stringRef;
    }

    public String getString(int stringRef) {
        if (stringRef == 0) {
            return null;
        }

        validateStringRef(stringRef);

        String cached = stringRef < stringsByRef.size() ? stringsByRef.get(stringRef) : null;
        if (cached != null) {
            return cached;
        }

        long recordOffset = stringRecordOffset(stringRef);
        MemorySegment record = stringRecordHeap.segment();

        long byteOffset = record.get(Layouts.I64, recordOffset + Layouts.STRING_BYTE_OFFSET);
        int byteLength = record.get(Layouts.I32, recordOffset + Layouts.STRING_BYTE_LENGTH);

        byte[] bytes = new byte[byteLength];
        MemorySegment data = stringByteHeap.segment();

        for (int i = 0; i < byteLength; i++) {
            bytes[i] = data.get(Layouts.I8, byteOffset + i);
        }

        String value = new String(bytes, StandardCharsets.UTF_8);

        while (stringsByRef.size() <= stringRef) {
            stringsByRef.add(null);
        }
        stringsByRef.set(stringRef, value);

        return value;
    }

    public int createArray(ValueKind kind, int length) {
        requireOpen();
        Objects.requireNonNull(kind, "kind");

        if (length < 0) {
            throw new IllegalArgumentException("length must be non-negative");
        }

        int typeCode = arrayTypeFor(kind);
        int fsRef = createFeatureStructure(typeCode, 0);

        long headerOffset = arrayHeap.allocate(Layouts.ARRAY_HEADER_SIZE, 8);
        long dataOffset = length == 0
                ? NULL_OFFSET
                : arrayHeap.allocate(length * (long) kind.byteWidth(), alignmentFor(kind));

        writeArrayHeader(headerOffset, typeCode, kind, length, length, dataOffset);
        setAuxOffset(fsRef, headerOffset);

        return fsRef;
    }

    public void resizeArray(int arrayFsRef, int newLength) {
        if (newLength < 0) {
            throw new IllegalArgumentException("newLength must be non-negative");
        }

        long header = arrayHeaderOffset(arrayFsRef);
        ValueKind kind = arrayKind(arrayFsRef);

        int oldLength = arrayLength(arrayFsRef);
        int capacity = arrayCapacity(arrayFsRef);

        long dataOffset = arrayHeap.segment().get(
                Layouts.I64,
                header + Layouts.ARRAY_DATA_OFFSET
        );

        if (newLength > capacity) {
            int newCapacity = Math.max(1, capacity);
            while (newCapacity < newLength) {
                newCapacity = Math.multiplyExact(newCapacity, 2);
            }

            long newDataOffset = arrayHeap.allocate(
                    newCapacity * (long) kind.byteWidth(),
                    alignmentFor(kind)
            );

            if (dataOffset != NULL_OFFSET && oldLength > 0) {
                MemorySegment.copy(
                        arrayHeap.segment(),
                        dataOffset,
                        arrayHeap.segment(),
                        newDataOffset,
                        oldLength * (long) kind.byteWidth()
                );
            }

            dataOffset = newDataOffset;
            arrayHeap.segment().set(Layouts.I64, header + Layouts.ARRAY_DATA_OFFSET, dataOffset);
            arrayHeap.segment().set(Layouts.I32, header + Layouts.ARRAY_CAPACITY, newCapacity);
        }

        if (newLength > oldLength) {
            zeroArrayRange(dataOffset, kind, oldLength, newLength - oldLength);
        }

        arrayHeap.segment().set(Layouts.I32, header + Layouts.ARRAY_LENGTH, newLength);
    }

    public int createNonEmptyList(ValueKind kind, long encodedHeadSlot, int tailFsRef) {
        int typeCode = nonEmptyListTypeFor(kind);
        int fsRef = createFeatureStructure(typeCode, 2);

        if (tailFsRef != 0) {
            validateFsRef(tailFsRef);
        }

        setRawFeatureSlot(fsRef, LIST_FEAT_HEAD, encodedHeadSlot);
        setFsRefFeature(fsRef, LIST_FEAT_TAIL, tailFsRef);

        return fsRef;
    }

    public int copyFeatureStructure(int fsRef) {
        validateFsRef(fsRef);

        int typeCode = getTypeCode(fsRef);
        int featureCount = getFeatureCount(fsRef);
        int copy = createFeatureStructure(typeCode, featureCount);

        setFlags(copy, getFlags(fsRef));

        for (int i = 0; i < featureCount; i++) {
            setRawFeatureSlot(copy, i, getRawFeatureSlot(fsRef, i));
        }

        if (isArrayType(typeCode)) {
            cloneArrayAux(fsRef, copy);
        } else {
            setAuxOffset(copy, getAuxOffset(fsRef));
        }

        return copy;
    }

    public Mark mark() {
        return new Mark(
                fsCount,
                stringCount,
                fsHeap.used(),
                featureHeap.used(),
                arrayHeap.used(),
                stringRecordHeap.used(),
                stringByteHeap.used()
        );
    }

    public synchronized void resetToMark(Mark mark) {
        Objects.requireNonNull(mark, "mark");

        if (mark.fsCount < 0 || mark.fsCount > fsCount
                || mark.stringCount < 0 || mark.stringCount > stringCount) {
            throw new IllegalArgumentException("mark is not from this backend state");
        }

        fsCount = mark.fsCount;
        stringCount = mark.stringCount;

        fsHeap.setUsed(mark.fsHeapUsed);
        featureHeap.setUsed(mark.featureHeapUsed);
        arrayHeap.setUsed(mark.arrayHeapUsed);
        stringRecordHeap.setUsed(mark.stringRecordHeapUsed);
        stringByteHeap.setUsed(mark.stringByteHeapUsed);

        while (stringsByRef.size() > stringCount + 1) {
            String removed = stringsByRef.remove(stringsByRef.size() - 1);
            if (removed != null) {
                internedStrings.remove(removed);
            }
        }

        internedStrings.entrySet().removeIf(e -> e.getValue() > stringCount);
        indexes.clear();
    }

    public synchronized void clearCas() {
        fsHeap.clear();
        featureHeap.clear();
        arrayHeap.clear();
        stringRecordHeap.clear();
        stringByteHeap.clear();

        internedStrings.clear();
        stringsByRef.clear();
        stringsByRef.add(null);

        indexes.clear();

        fsCount = 0;
        stringCount = 0;
    }

    @Override
    public synchronized void close() {
        if (!closed) {
            closed = true;
            if (ownsArena) {
                arena.close();
            }
        }
    }

    private void cloneArrayAux(int sourceFsRef, int targetFsRef) {
        long sourceHeader = arrayHeaderOffset(sourceFsRef);
        ValueKind kind = arrayKind(sourceFsRef);

        int length = arrayLength(sourceFsRef);
        int capacity = arrayCapacity(sourceFsRef);

        long sourceData = arrayHeap.segment().get(
                Layouts.I64,
                sourceHeader + Layouts.ARRAY_DATA_OFFSET
        );

        long targetHeader = arrayHeap.allocate(Layouts.ARRAY_HEADER_SIZE, 8);
        long targetData = capacity == 0
                ? NULL_OFFSET
                : arrayHeap.allocate(capacity * (long) kind.byteWidth(), alignmentFor(kind));

        writeArrayHeader(targetHeader, arrayTypeFor(kind), kind, length, capacity, targetData);

        if (length > 0) {
            MemorySegment.copy(
                    arrayHeap.segment(),
                    sourceData,
                    arrayHeap.segment(),
                    targetData,
                    length * (long) kind.byteWidth()
            );
        }

        setAuxOffset(targetFsRef, targetHeader);
    }

    private long recordOffset(int fsRef) {
        validateFsRef(fsRef);
        return (fsRef - 1L) * Layouts.FS_RECORD_SIZE;
    }

    private long stringRecordOffset(int stringRef) {
        return (stringRef - 1L) * Layouts.STRING_RECORD_SIZE;
    }

    private long featureSlotOffset(int fsRef, int featureIndex) {
        long record = recordOffset(fsRef);

        int count = fsHeap.segment().get(
                Layouts.I32,
                record + Layouts.FS_FEATURE_COUNT
        );

        if (featureIndex < 0 || featureIndex >= count) {
            throw new IndexOutOfBoundsException(
                    "featureIndex=" + featureIndex + ", featureCount=" + count
            );
        }

        long featureOffset = fsHeap.segment().get(
                Layouts.I64,
                record + Layouts.FS_FEATURE_OFFSET
        );

        if (featureOffset == NULL_OFFSET) {
            throw new IllegalStateException("FS has no feature slots: " + fsRef);
        }

        return featureOffset + featureIndex * (long) Layouts.SLOT_WIDTH;
    }

    private long arrayHeaderOffset(int arrayFsRef) {
        validateFsRef(arrayFsRef);

        if (!isArrayType(getTypeCode(arrayFsRef))) {
            throw new IllegalArgumentException("Not an array FS: " + arrayFsRef);
        }

        long aux = getAuxOffset(arrayFsRef);

        if (aux == NULL_OFFSET) {
            throw new IllegalStateException("Array has no auxiliary header: " + arrayFsRef);
        }

        return aux;
    }

    private long arrayElementOffset(int arrayFsRef, int index, ValueKind kind) {
        long header = arrayHeaderOffset(arrayFsRef);

        int length = arrayHeap.segment().get(
                Layouts.I32,
                header + Layouts.ARRAY_LENGTH
        );

        if (index < 0 || index >= length) {
            throw new IndexOutOfBoundsException("index=" + index + ", length=" + length);
        }

        long dataOffset = arrayHeap.segment().get(
                Layouts.I64,
                header + Layouts.ARRAY_DATA_OFFSET
        );

        if (dataOffset == NULL_OFFSET) {
            throw new IndexOutOfBoundsException("empty array");
        }

        return dataOffset + index * (long) kind.byteWidth();
    }

    private void writeArrayHeader(
            long headerOffset,
            int typeCode,
            ValueKind kind,
            int length,
            int capacity,
            long dataOffset
    ) {
        MemorySegment arrays = arrayHeap.segment();
        arrays.set(Layouts.I32, headerOffset + Layouts.ARRAY_TYPE_CODE, typeCode);
        arrays.set(Layouts.I32, headerOffset + Layouts.ARRAY_VALUE_KIND, kind.code());
        arrays.set(Layouts.I32, headerOffset + Layouts.ARRAY_LENGTH, length);
        arrays.set(Layouts.I32, headerOffset + Layouts.ARRAY_CAPACITY, capacity);
        arrays.set(Layouts.I64, headerOffset + Layouts.ARRAY_DATA_OFFSET, dataOffset);
    }

    private void zeroArrayRange(long dataOffset, ValueKind kind, int start, int length) {
        if (length <= 0) {
            return;
        }

        long from = dataOffset + start * (long) kind.byteWidth();
        long bytes = length * (long) kind.byteWidth();

        for (long i = 0; i < bytes; i++) {
            arrayHeap.segment().set(Layouts.I8, from + i, (byte) 0);
        }
    }

    private void checkArrayKind(int arrayFsRef, ValueKind expected) {
        ValueKind actual = arrayKind(arrayFsRef);

        if (actual != expected) {
            throw new IllegalArgumentException(
                    "Expected array kind " + expected + " but was " + actual
            );
        }
    }

    private void checkListKind(int listFsRef, ValueKind expected) {
        int typeCode = getTypeCode(listFsRef);

        if (isEmptyListType(typeCode)) {
            throw new IllegalArgumentException("Empty list has no head: " + listFsRef);
        }

        ValueKind actual = listElementKindFor(typeCode);

        if (actual != expected) {
            throw new IllegalArgumentException(
                    "Expected list kind " + expected + " but was " + actual
            );
        }
    }

    private void validateFsRef(int fsRef) {
        if (fsRef <= 0 || fsRef > fsCount) {
            throw new IllegalArgumentException("Invalid FS ref: " + fsRef + ", fsCount=" + fsCount);
        }
    }

    private void validateStringRef(int stringRef) {
        if (stringRef < 0 || stringRef > stringCount) {
            throw new IllegalArgumentException(
                    "Invalid string ref: " + stringRef + ", stringCount=" + stringCount
            );
        }
    }

    private void requireOpen() {
        if (closed) {
            throw new IllegalStateException("backend is closed");
        }
    }

    private static int alignmentFor(ValueKind kind) {
        return switch (kind) {
            case BOOLEAN, BYTE -> 1;
            case SHORT -> 2;
            case INT, FLOAT, STRING, FS_REF -> 4;
            case LONG, DOUBLE -> 8;
        };
    }

    private static boolean isArrayType(int typeCode) {
        return typeCode >= TYPE_BOOLEAN_ARRAY && typeCode <= TYPE_FS_ARRAY;
    }

    private static int arrayTypeFor(ValueKind kind) {
        return switch (kind) {
            case BOOLEAN -> TYPE_BOOLEAN_ARRAY;
            case BYTE -> TYPE_BYTE_ARRAY;
            case SHORT -> TYPE_SHORT_ARRAY;
            case INT -> TYPE_INT_ARRAY;
            case LONG -> TYPE_LONG_ARRAY;
            case FLOAT -> TYPE_FLOAT_ARRAY;
            case DOUBLE -> TYPE_DOUBLE_ARRAY;
            case STRING -> TYPE_STRING_ARRAY;
            case FS_REF -> TYPE_FS_ARRAY;
        };
    }

    private static int emptyListTypeFor(ValueKind kind) {
        return switch (kind) {
            case INT -> TYPE_EMPTY_INTEGER_LIST;
            case FLOAT -> TYPE_EMPTY_FLOAT_LIST;
            case STRING -> TYPE_EMPTY_STRING_LIST;
            case FS_REF -> TYPE_EMPTY_FS_LIST;
            default -> throw new IllegalArgumentException(
                    "UIMA built-in lists are only INT, FLOAT, STRING, and FS_REF: " + kind
            );
        };
    }

    private static int nonEmptyListTypeFor(ValueKind kind) {
        return switch (kind) {
            case INT -> TYPE_NON_EMPTY_INTEGER_LIST;
            case FLOAT -> TYPE_NON_EMPTY_FLOAT_LIST;
            case STRING -> TYPE_NON_EMPTY_STRING_LIST;
            case FS_REF -> TYPE_NON_EMPTY_FS_LIST;
            default -> throw new IllegalArgumentException(
                    "UIMA built-in lists are only INT, FLOAT, STRING, and FS_REF: " + kind
            );
        };
    }

    private static ValueKind listElementKindFor(int typeCode) {
        return switch (typeCode) {
            case TYPE_EMPTY_INTEGER_LIST, TYPE_NON_EMPTY_INTEGER_LIST -> ValueKind.INT;
            case TYPE_EMPTY_FLOAT_LIST, TYPE_NON_EMPTY_FLOAT_LIST -> ValueKind.FLOAT;
            case TYPE_EMPTY_STRING_LIST, TYPE_NON_EMPTY_STRING_LIST -> ValueKind.STRING;
            case TYPE_EMPTY_FS_LIST, TYPE_NON_EMPTY_FS_LIST -> ValueKind.FS_REF;
            default -> throw new IllegalArgumentException("Not a built-in list type: " + typeCode);
        };
    }

    private static boolean isEmptyListType(int typeCode) {
        return typeCode == TYPE_EMPTY_INTEGER_LIST
                || typeCode == TYPE_EMPTY_FLOAT_LIST
                || typeCode == TYPE_EMPTY_STRING_LIST
                || typeCode == TYPE_EMPTY_FS_LIST;
    }

    private static final class SegmentHeap {
        private final Arena arena;
        private final String name;
        private final long defaultAlignment;

        private MemorySegment segment;
        private long capacity;
        private long used;

        SegmentHeap(Arena arena, String name, long initialCapacity, long defaultAlignment) {
            this.arena = arena;
            this.name = name;

            if (defaultAlignment <= 0 || Long.bitCount(defaultAlignment) != 1) {
                throw new IllegalArgumentException("defaultAlignment must be a positive power of two");
            }

            this.defaultAlignment = defaultAlignment;
            this.capacity = Math.max(defaultAlignment, alignUp(initialCapacity, defaultAlignment));
            this.segment = arena.allocate(capacity, defaultAlignment);
        }

        synchronized long allocate(long byteSize, long byteAlignment) {
            if (byteSize < 0) {
                throw new IllegalArgumentException("byteSize must be non-negative");
            }

            if (byteAlignment <= 0 || Long.bitCount(byteAlignment) != 1) {
                throw new IllegalArgumentException("byteAlignment must be a positive power of two");
            }

            long aligned = alignUp(used, byteAlignment);
            long end = Math.addExact(aligned, byteSize);

            ensureCapacity(end);
            used = end;

            return aligned;
        }

        synchronized void ensureCapacity(long minCapacity) {
            if (minCapacity <= capacity) {
                return;
            }

            long newCapacity = capacity;
            while (newCapacity < minCapacity) {
                newCapacity = Math.multiplyExact(newCapacity, 2L);
            }

            MemorySegment newSegment = arena.allocate(newCapacity, defaultAlignment);
            MemorySegment.copy(segment, 0, newSegment, 0, used);

            segment = newSegment;
            capacity = newCapacity;
        }

        synchronized void setUsed(long newUsed) {
            if (newUsed < 0 || newUsed > used) {
                throw new IllegalArgumentException("invalid used mark for " + name + ": " + newUsed);
            }

            used = newUsed;
        }

        synchronized void clear() {
            used = 0L;
        }

        MemorySegment segment() {
            return segment;
        }

        synchronized long used() {
            return used;
        }

        static long alignUp(long value, long alignment) {
            long mask = alignment - 1L;
            return (value + mask) & ~mask;
        }
    }

    private static final class IntVector {
        private final SegmentHeap heap;

        private long dataOffset;
        private int size;
        private int capacity;

        IntVector(Arena arena, String name, int initialCapacity) {
            this.capacity = Math.max(4, initialCapacity);
            this.heap = new SegmentHeap(
                    arena,
                    name,
                    capacity * (long) Integer.BYTES,
                    Integer.BYTES
            );
            this.dataOffset = heap.allocate(capacity * (long) Integer.BYTES, Integer.BYTES);
        }

        int size() {
            return size;
        }

        int get(int index) {
            check(index);
            return heap.segment().get(
                    Layouts.I32,
                    dataOffset + index * (long) Integer.BYTES
            );
        }

        void set(int index, int value) {
            check(index);
            heap.segment().set(
                    Layouts.I32,
                    dataOffset + index * (long) Integer.BYTES,
                    value
            );
        }

        boolean contains(int value) {
            for (int i = 0; i < size; i++) {
                if (get(i) == value) {
                    return true;
                }
            }
            return false;
        }

        void add(int value) {
            ensure(size + 1);
            heap.segment().set(
                    Layouts.I32,
                    dataOffset + size * (long) Integer.BYTES,
                    value
            );
            size++;
        }

        boolean removeValue(int value) {
            for (int i = 0; i < size; i++) {
                if (get(i) == value) {
                    int last = get(size - 1);
                    set(i, last);
                    size--;
                    return true;
                }
            }
            return false;
        }

        int[] toArray() {
            int[] out = new int[size];

            for (int i = 0; i < size; i++) {
                out[i] = get(i);
            }

            return out;
        }

        void clear() {
            size = 0;
        }

        private void ensure(int min) {
            if (min <= capacity) {
                return;
            }

            int newCapacity = capacity;
            while (newCapacity < min) {
                newCapacity = Math.multiplyExact(newCapacity, 2);
            }

            long newOffset = heap.allocate(
                    newCapacity * (long) Integer.BYTES,
                    Integer.BYTES
            );

            MemorySegment.copy(
                    heap.segment(),
                    dataOffset,
                    heap.segment(),
                    newOffset,
                    size * (long) Integer.BYTES
            );

            dataOffset = newOffset;
            capacity = newCapacity;
        }

        private void check(int index) {
            if (index < 0 || index >= size) {
                throw new IndexOutOfBoundsException("index=" + index + ", size=" + size);
            }
        }
    }

    private static final class IndexRepository {
        private final Arena arena;
        private final Map<Integer, IntVector> byExactType = new HashMap<>();

        IndexRepository(Arena arena) {
            this.arena = arena;
        }

        void add(int typeCode, int fsRef) {
            IntVector vector = byExactType.computeIfAbsent(
                    typeCode,
                    t -> new IntVector(arena, "idx-" + t, 16)
            );

            if (!vector.contains(fsRef)) {
                vector.add(fsRef);
            }
        }

        boolean remove(int typeCode, int fsRef) {
            IntVector vector = byExactType.get(typeCode);
            return vector != null && vector.removeValue(fsRef);
        }

        int[] snapshot(int typeCode) {
            IntVector vector = byExactType.get(typeCode);
            return vector == null ? new int[0] : vector.toArray();
        }

        void clear() {
            for (IntVector vector : byExactType.values()) {
                vector.clear();
            }
            byExactType.clear();
        }
    }
